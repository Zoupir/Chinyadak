function sliceCallExpression(text: string, start: number, maxChars = 5000) {
  const open = text.indexOf('(', start);
  if (open < 0) return text.slice(start, start + maxChars);
  let depth = 0;
  let quote = '';
  let escaped = false;
  const limit = Math.min(text.length, start + maxChars);
  for (let index = open; index < limit; index += 1) {
    const char = text[index];
    if (quote) {
      if (escaped) {
        escaped = false;
        continue;
      }
      if (char === '\\') {
        escaped = true;
        continue;
      }
      if (char === quote) quote = '';
      continue;
    }
    if (char === '"' || char === "'") {
      quote = char;
      continue;
    }
    if (char === '(') depth += 1;
    else if (char === ')') {
      depth -= 1;
      if (depth === 0) return text.slice(start, index + 1);
    }
  }
  return text.slice(start, limit);
}
