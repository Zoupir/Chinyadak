import React, { useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { Check, FileImage, Folder, FolderOpen, Loader2, Pencil, Search, Upload, X } from 'lucide-react';
import {
  listMediaLibrary,
  MediaLibraryItem,
  MediaSeoMeta,
  MediaUploadError,
  updateMediaSeo,
  uploadImage
} from '../../api/media';

interface MediaPickerModalProps {
  isOpen: boolean;
  onClose: () => void;
  onSelect: (url: string, item?: MediaLibraryItem) => void;
  category?: string;
  initialUrl?: string;
  title?: string;
  allowUrl?: boolean;
}

type Tab = 'library' | 'upload' | 'url';
const PAGE_SIZE = 80;

export const MediaPickerModal: React.FC<MediaPickerModalProps> = ({
  isOpen, onClose, onSelect, category = 'general', initialUrl = '', title = 'انتخاب رسانه', allowUrl = true
}) => {
  const [tab, setTab] = useState<Tab>('library');
  const [items, setItems] = useState<MediaLibraryItem[]>([]);
  const [categories, setCategories] = useState<string[]>([]);
  const [years, setYears] = useState<string[]>([]);
  const [months, setMonths] = useState<string[]>([]);
  const [filterCategory, setFilterCategory] = useState('all');
  const [filterYear, setFilterYear] = useState('all');
  const [filterMonth, setFilterMonth] = useState('all');
  const [query, setQuery] = useState('');
  const [selected, setSelected] = useState<MediaLibraryItem | null>(null);
  const [loading, setLoading] = useState(false);
  const [loadingMore, setLoadingMore] = useState(false);
  const [total, setTotal] = useState(0);
  const [hasMore, setHasMore] = useState(false);
  const [nextOffset, setNextOffset] = useState<number | null>(null);
  const [loadError, setLoadError] = useState('');
  const [uploading, setUploading] = useState(false);
  const [uploadError, setUploadError] = useState('');
  const [directUrl, setDirectUrl] = useState('');
  const [pendingSeoPath, setPendingSeoPath] = useState('');
  const [seoDraft, setSeoDraft] = useState<MediaSeoMeta>({ alt:'', title:'', caption:'', description:'' });
  const [savingSeo, setSavingSeo] = useState(false);
  const [editingSeo, setEditingSeo] = useState(false);
  const [metadataError, setMetadataError] = useState('');
  const fileRef = useRef<HTMLInputElement>(null);
  const autoSelectedUrlRef = useRef('');

  const load = async (append = false, force = false) => {
    append ? setLoadingMore(true) : setLoading(true);
    setLoadError('');
    try {
      const result = await listMediaLibrary({
        q: query.trim() || undefined,
        category: filterCategory,
        year: filterYear,
        month: filterMonth,
        kind: 'image',
        limit: PAGE_SIZE,
        offset: append ? (nextOffset || items.length) : 0,
        refresh: force
      });
      setItems(current => append ? [...current, ...(result.items || [])] : (result.items || []));
      setCategories(result.categories || []);
      setYears(result.years || []);
      setMonths(result.months || []);
      setTotal(result.total || 0);
      setHasMore(Boolean(result.hasMore));
      setNextOffset(result.nextOffset ?? null);
    } catch (error) {
      if (!append) setItems([]);
      setLoadError(error instanceof Error ? error.message : 'دریافت فهرست رسانه‌ها انجام نشد.');
    } finally {
      setLoading(false); setLoadingMore(false);
    }
  };

  useEffect(() => {
    if (!isOpen) return;
    const timer = window.setTimeout(() => void load(false), 250);
    return () => window.clearTimeout(timer);
  }, [isOpen, filterCategory, filterYear, filterMonth, query]);

  useEffect(() => {
    if (!isOpen) {
      setSelected(null); setQuery(''); setUploadError(''); setDirectUrl('');
      setPendingSeoPath(''); setEditingSeo(false); setMetadataError('');
      autoSelectedUrlRef.current = '';
    }
  }, [isOpen]);

  useEffect(() => {
    if (!isOpen || !initialUrl) return;
    setTab('library'); setFilterCategory('all'); setFilterYear('all'); setFilterMonth('all');
  }, [isOpen, initialUrl]);

  useEffect(() => {
    if (!isOpen || !initialUrl || !items.length || autoSelectedUrlRef.current === initialUrl) return;
    const normalize = (value: string) => {
      try { const parsed = new URL(value, window.location.origin); return parsed.origin === window.location.origin ? parsed.pathname : value; }
      catch { return value; }
    };
    const match = items.find(item => normalize(item.url) === normalize(initialUrl));
    if (match) { setSelected(match); autoSelectedUrlRef.current = initialUrl; }
  }, [isOpen, initialUrl, items]);

  const uploadFiles = async (files: FileList | File[]) => {
    const list = Array.from(files); if (!list.length) return;
    setUploading(true); setUploadError('');
    try {
      let lastUploaded: Awaited<ReturnType<typeof uploadImage>> | null = null;
      for (const file of list) lastUploaded = await uploadImage(file, category);
      const refreshed = await listMediaLibrary({ category, kind:'image', limit: PAGE_SIZE, refresh:true });
      setItems(refreshed.items || []); setCategories(refreshed.categories || []); setYears(refreshed.years || []); setMonths(refreshed.months || []);
      setTotal(refreshed.total || 0); setHasMore(Boolean(refreshed.hasMore)); setNextOffset(refreshed.nextOffset ?? null);
      setFilterCategory(category); setFilterYear('all'); setFilterMonth('all');
      if (list.length === 1 && lastUploaded) {
        const item = refreshed.items.find(media => media.relativePath === lastUploaded?.relativePath || media.url === lastUploaded?.url) || lastUploaded;
        setSelected(item);
        setPendingSeoPath(item.relativePath);
        setSeoDraft({ alt:item.seo?.alt || '', title:item.seo?.title || '', caption:item.seo?.caption || '', description:item.seo?.description || '' });
        setTab('upload');
      } else { setSelected(null); setPendingSeoPath(''); setTab('library'); }
    } catch (error) {
      setUploadError(error instanceof MediaUploadError
        ? error.code === 'MEDIA_FILENAME_EXISTS'
          ? 'فایلی دقیقاً با همین نام در پوشه همین ماه وجود دارد. نام فایل را تغییر دهید یا فایل قبلی را حذف کنید.'
          : `آپلود انجام نشد: ${error.code}`
        : 'آپلود رسانه انجام نشد.');
    } finally { setUploading(false); if (fileRef.current) fileRef.current.value = ''; }
  };

  const saveUploadedSeoAndSelect = async () => {
    if (!selected || selected.relativePath !== pendingSeoPath) return;
    setSavingSeo(true); setUploadError('');
    try {
      const seo = await updateMediaSeo(selected.relativePath, seoDraft);
      const updated = { ...selected, seo };
      setSelected(updated); setItems(current => current.map(item => item.relativePath === updated.relativePath ? updated : item));
      onSelect(updated.url, updated); onClose();
    } catch (error) {
      setUploadError(error instanceof MediaUploadError ? `ذخیره اطلاعات ناموفق بود: ${error.code}` : 'ذخیره اطلاعات ناموفق بود.');
    } finally { setSavingSeo(false); }
  };

  const openSelectedSeoEditor = () => {
    if (!selected) return;
    setSeoDraft({ alt:selected.seo?.alt || '', title:selected.seo?.title || '', caption:selected.seo?.caption || '', description:selected.seo?.description || '' });
    setMetadataError(''); setEditingSeo(true);
  };
  const saveSelectedSeo = async () => {
    if (!selected) return;
    setSavingSeo(true); setMetadataError('');
    try {
      const seo = await updateMediaSeo(selected.relativePath, seoDraft);
      const updated = { ...selected, seo };
      setSelected(updated); setItems(current => current.map(item => item.relativePath === updated.relativePath ? updated : item)); setEditingSeo(false);
    } catch (error) {
      setMetadataError(error instanceof MediaUploadError ? `ذخیره اطلاعات ناموفق بود: ${error.code}` : 'ذخیره اطلاعات ناموفق بود.');
    } finally { setSavingSeo(false); }
  };

  if (!isOpen) return null;
  return createPortal(
    <div className="fixed inset-0 z-[99999] bg-black/65 backdrop-blur-sm p-1 sm:p-3 lg:p-6 flex items-center justify-center" onClick={onClose}>
      <div role="dialog" aria-modal="true" aria-label={title} className="w-[calc(100vw-0.5rem)] sm:w-[96vw] max-w-[1440px] h-[calc(100dvh-0.5rem)] sm:h-[88dvh] max-h-[820px] bg-white rounded-xl sm:rounded-2xl shadow-2xl overflow-hidden flex flex-col" onClick={event => event.stopPropagation()}>
        <header className="min-h-14 px-3 sm:px-6 border-b flex items-center justify-between gap-3 shrink-0"><div><h2 className="font-black text-base">{title}</h2><p className="text-[10px] text-neutral-500">کتابخانه تصویر با نسخه‌های بهینه WebP</p></div><button onClick={onClose} className="w-9 h-9 grid place-items-center rounded-xl bg-neutral-100"><X className="w-4 h-4" /></button></header>
        <div className="px-2 sm:px-6 border-b flex items-center gap-1 shrink-0 overflow-x-auto">{([['library','کتابخانه رسانه'],['upload','بارگذاری فایل جدید'],...(allowUrl ? [['url','لینک مستقیم']] : [])] as Array<[Tab,string]>).map(([id,label]) => <button key={id} onClick={() => setTab(id)} className={`px-4 py-3 text-xs font-bold border-b-2 whitespace-nowrap ${tab === id ? 'border-blue-600 text-blue-700' : 'border-transparent text-neutral-500'}`}>{label}</button>)}</div>

        {tab === 'library' && <div className="min-h-0 flex-1 grid grid-cols-1 lg:grid-cols-[210px_minmax(0,1fr)] 2xl:grid-cols-[210px_minmax(0,1fr)_270px]">
          <aside className="border-l bg-neutral-50 p-3 overflow-y-auto hidden lg:block"><strong className="text-[11px]">پوشه‌ها</strong><button onClick={() => {setFilterCategory('all');setFilterYear('all');setFilterMonth('all');}} className={`mt-2 w-full p-2 rounded-lg text-right text-[10px] font-bold flex gap-2 ${filterCategory === 'all' ? 'bg-blue-600 text-white' : ''}`}><FolderOpen className="w-4 h-4" /> همه</button><div className="mt-2">{categories.map(cat => <div key={cat}><button onClick={() => {setFilterCategory(cat);setFilterYear('all');setFilterMonth('all');}} className={`w-full p-2 rounded-lg text-right text-[10px] font-bold flex gap-2 ${filterCategory === cat && filterYear === 'all' ? 'bg-white text-blue-700' : ''}`}><Folder className="w-4 h-4" />{cat}</button>{filterCategory === cat && <div className="mr-5 border-r pr-2">{years.map(y => <div key={y}><button onClick={() => {setFilterYear(y);setFilterMonth('all');}} className={`w-full p-1.5 text-right text-[9px] ${filterYear === y ? 'font-bold text-blue-700' : ''}`}>{y}</button>{filterYear === y && <div className="mr-3">{months.map(m => <button key={m} onClick={() => setFilterMonth(m)} className={`block w-full p-1.5 text-right text-[9px] rounded ${filterMonth === m ? 'bg-blue-600 text-white' : ''}`}>ماه {m}</button>)}</div>}</div>)}</div>}</div>)}</div></aside>
          <main className="min-w-0 min-h-0 flex flex-col"><div className="p-3 border-b flex flex-wrap items-center gap-2 shrink-0"><div className="relative flex-1 min-w-[200px]"><Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-neutral-400" /><input value={query} onChange={e => setQuery(e.target.value)} className="w-full h-10 pr-10 pl-3 border rounded-xl text-xs" placeholder="جستجو در کل کتابخانه..." /></div><select value={filterCategory} onChange={e => {setFilterCategory(e.target.value);setFilterYear('all');setFilterMonth('all');}} className="lg:hidden h-10 border rounded-xl text-[10px]"><option value="all">همه پوشه‌ها</option>{categories.map(cat => <option key={cat}>{cat}</option>)}</select><span className="text-[10px] text-neutral-400">{total.toLocaleString('fa-IR')} تصویر</span></div>
            <div className="flex-1 overflow-y-auto p-3">{loading ? <div className="h-full grid place-items-center"><Loader2 className="w-7 h-7 animate-spin" /></div> : items.length ? <><div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 xl:grid-cols-7 2xl:grid-cols-5 gap-2">{items.map(item => <button key={item.relativePath} onClick={() => setSelected(item)} className={`relative rounded-xl border overflow-hidden bg-neutral-50 text-right ${selected?.relativePath === item.relativePath ? 'border-blue-600 ring-2 ring-blue-100' : 'border-neutral-200'}`}>{selected?.relativePath === item.relativePath && <span className="absolute top-1 left-1 z-10 w-5 h-5 rounded-full bg-blue-600 text-white grid place-items-center"><Check className="w-3 h-3" /></span>}<div className="aspect-square p-2 bg-white"><img src={item.thumbnailUrl || item.url} alt={item.seo?.alt || item.filename} className="w-full h-full object-contain" loading="lazy" /></div><div className="p-2"><strong className="block text-[8px] truncate">{item.seo?.title || item.filename}</strong></div></button>)}</div>{hasMore && <button disabled={loadingMore} onClick={() => void load(true)} className="mt-4 w-full py-2.5 border rounded-xl text-xs font-bold">{loadingMore ? 'در حال دریافت…' : `موارد بیشتر (${Math.max(0,total-items.length).toLocaleString('fa-IR')} باقی‌مانده)`}</button>}</> : <div className="h-full grid place-items-center text-xs text-neutral-400">{loadError || 'رسانه‌ای پیدا نشد.'}</div>}</div>
            {selected && <div className="2xl:hidden p-3 border-t flex gap-2 items-center"><img src={selected.thumbnailUrl || selected.url} alt={selected.seo?.alt || ''} className="w-12 h-12 object-contain border rounded" /><span className="flex-1 truncate text-[10px] font-bold">{selected.seo?.title || selected.filename}</span><button onClick={openSelectedSeoEditor} className="px-3 py-2 border rounded-lg text-[10px]"><Pencil className="w-3 h-3 inline" /> ویرایش</button><button onClick={() => {onSelect(selected.url,selected);onClose();}} className="px-3 py-2 bg-blue-600 text-white rounded-lg text-[10px] font-bold">انتخاب</button></div>}
          </main>
          <aside className="border-r p-4 overflow-y-auto hidden 2xl:block">{selected ? <div className="space-y-4"><div className="aspect-video rounded-xl border bg-neutral-50 p-2"><img src={selected.thumbnailUrl || selected.url} alt={selected.seo?.alt || ''} className="w-full h-full object-contain" /></div><div className="text-[10px]"><strong className="block break-all">{selected.seo?.title || selected.filename}</strong><span className="text-neutral-400">ALT: {selected.seo?.alt || 'تنظیم نشده'}</span></div><button onClick={openSelectedSeoEditor} className="w-full py-2.5 border rounded-xl text-xs font-bold">ویرایش اطلاعات</button><button onClick={() => {onSelect(selected.url,selected);onClose();}} className="w-full py-2.5 bg-blue-600 text-white rounded-xl text-xs font-black">استفاده از این تصویر</button></div> : <div className="h-full grid place-items-center text-[10px] text-neutral-400">یک تصویر را انتخاب کنید.</div>}</aside>
        </div>}

        {tab === 'upload' && <div className="flex-1 overflow-y-auto p-4 sm:p-8 grid place-items-center">{pendingSeoPath && selected?.relativePath === pendingSeoPath ? <div className="w-full max-w-3xl space-y-4"><div className="flex gap-3 p-3 border bg-emerald-50 rounded-2xl"><img src={selected.thumbnailUrl || selected.url} alt={seoDraft.alt} className="w-20 h-20 object-contain border bg-white rounded-xl" /><div><strong className="text-xs">تصویر بارگذاری شد</strong><p className="text-[10px] text-neutral-500">اطلاعات سئو را پیش از استفاده ثبت کنید.</p></div></div><div className="grid sm:grid-cols-2 gap-3"><label className="text-[10px] font-bold">ALT<input value={seoDraft.alt} onChange={e => setSeoDraft({...seoDraft,alt:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs font-normal" /></label><label className="text-[10px] font-bold">عنوان<input value={seoDraft.title} onChange={e => setSeoDraft({...seoDraft,title:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs font-normal" /></label><label className="text-[10px] font-bold">توضیح کوتاه<textarea rows={2} value={seoDraft.caption} onChange={e => setSeoDraft({...seoDraft,caption:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs font-normal" /></label><label className="text-[10px] font-bold">توضیح کامل<textarea rows={2} value={seoDraft.description} onChange={e => setSeoDraft({...seoDraft,description:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs font-normal" /></label></div>{uploadError && <div className="p-3 bg-red-50 text-red-700 rounded-xl text-xs">{uploadError}</div>}<div className="flex gap-2"><button disabled={savingSeo} onClick={() => void saveUploadedSeoAndSelect()} className="flex-1 py-3 bg-emerald-600 text-white rounded-xl text-xs font-black">{savingSeo ? 'در حال ذخیره…' : 'ذخیره و استفاده'}</button><button onClick={() => {onSelect(selected.url,selected);onClose();}} className="px-4 py-3 border rounded-xl text-xs">استفاده بدون تغییر</button></div></div> : <div className="w-full max-w-xl"><input ref={fileRef} type="file" accept="image/*" multiple className="hidden" onChange={e => e.target.files && void uploadFiles(e.target.files)} /><button disabled={uploading} onClick={() => fileRef.current?.click()} className="w-full min-h-56 border-2 border-dashed rounded-3xl grid place-items-center text-center p-8"><div>{uploading ? <Loader2 className="w-9 h-9 animate-spin mx-auto" /> : <Upload className="w-9 h-9 mx-auto" />}<strong className="block mt-3 text-sm">انتخاب تصویر از دستگاه</strong><span className="text-[10px] text-neutral-500">نام فارسی فایل حفظ می‌شود و نسخه‌های WebP خودکار ساخته می‌شوند.</span></div></button>{uploadError && <div className="mt-3 p-3 bg-red-50 text-red-700 rounded-xl text-xs">{uploadError}</div>}</div>}</div>}

        {tab === 'url' && allowUrl && <div className="flex-1 p-6 grid place-items-center"><form onSubmit={e => {e.preventDefault();const value=directUrl.trim();if(value){onSelect(value);onClose();}}} className="w-full max-w-xl space-y-3"><label className="text-xs font-bold">لینک مستقیم تصویر<input dir="ltr" value={directUrl} onChange={e => setDirectUrl(e.target.value)} className="w-full mt-2 p-3 border rounded-xl text-left" placeholder="https://..." /></label><button className="w-full py-3 bg-blue-600 text-white rounded-xl text-xs font-black">استفاده از لینک</button></form></div>}
      </div>

      {editingSeo && selected && <div className="fixed inset-0 z-[100000] bg-black/55 p-3 flex items-center justify-center" onClick={() => setEditingSeo(false)}><div className="w-full max-w-lg bg-white rounded-2xl p-5 space-y-3" onClick={e => e.stopPropagation()}><div className="flex justify-between"><strong>ویرایش اطلاعات تصویر</strong><button onClick={() => setEditingSeo(false)}><X className="w-4 h-4" /></button></div><label className="block text-[10px] font-bold">ALT<input value={seoDraft.alt} onChange={e => setSeoDraft({...seoDraft,alt:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs" /></label><label className="block text-[10px] font-bold">عنوان<input value={seoDraft.title} onChange={e => setSeoDraft({...seoDraft,title:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs" /></label><label className="block text-[10px] font-bold">توضیح کوتاه<textarea rows={2} value={seoDraft.caption} onChange={e => setSeoDraft({...seoDraft,caption:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs" /></label><label className="block text-[10px] font-bold">توضیح کامل<textarea rows={3} value={seoDraft.description} onChange={e => setSeoDraft({...seoDraft,description:e.target.value})} className="w-full mt-1 p-2.5 border rounded-xl text-xs" /></label>{metadataError && <div className="text-xs text-red-700">{metadataError}</div>}<button disabled={savingSeo} onClick={() => void saveSelectedSeo()} className="w-full py-3 bg-emerald-600 text-white rounded-xl text-xs font-black">{savingSeo ? 'در حال ذخیره…' : 'ذخیره'}</button></div></div>}
    </div>,
    document.body
  );
};
